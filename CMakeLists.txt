cmake_minimum_required(VERSION 3.16)
if(DEFINED ENV{OPENP1_BUILD_VERSION} AND NOT "$ENV{OPENP1_BUILD_VERSION}" STREQUAL "")
    set(PROJECT_VER "$ENV{OPENP1_BUILD_VERSION}")
else()
    file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/VERSION" PROJECT_VER LIMIT_COUNT 1)
endif()
if(NOT PROJECT_VER MATCHES "^[0-9][0-9][0-9][0-9]\\.[0-9][0-9]\\.[0-9][0-9]$")
    message(FATAL_ERROR "Firmware version must use YYYY.MM.DD")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/VERSION")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(openp1_display)
